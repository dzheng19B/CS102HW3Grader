public static int maxValidWindowSum(int[] nums, int k) {
     int max = 0;
     int numNegatives = 0;
     int curr = 0;
     for(int i = 0; i<k; i++){
           if(nums[i] < 0){
                numNegatives++;
           }
           curr += nums[i];
     }
     if(numNegatives > 0){max = curr;}
     for(int i = 1; i<nums.length-k; i++){
           curr = curr + nums[i+k-1] - nums[i-1];
           if(nums[i-1] < 0){
                 numNegatives--;
           }
           if(nums[i+k] < 0){
                 numNegatives++;
           }
           max = numNegatives > 0 ? Math.max(curr, max) : max;
     }
     return max;
}
