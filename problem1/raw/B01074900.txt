public static int maxValidWindowSum(int[] nums, int k) {
    
    int size = k - 1; 
    int sum = 0; 
    int negativeCount = 0; 
    boolean firstValid = false; 

    //Check the first window
    for(int j = 0; j < k; j++){
          if(nums[j] < 0){
                     negativeCount++; 
           }
           sum += nums[j]; 

    }

    //Starting a valid window
    int greatestSum = 0; 
    if(negativeCount <= 1){
          firstValid = true; 
          greatestSum = sum; 
    }

    //Start Moving the window
    for(int i = size+1; i < nums.length; i++){
          

          //Check the leaving num
          sum -= nums[i-size-1];
          if(nums[i-size-1] < 0){
               negativeCount--; 
          }


         //Check incoming Num 
          sum += nums[i]; 
          if(nums[i] < 0){
                 negativeCount++; 
          }

           //not valid window
           if(negativeCount > 1){
                 continue;
           }

           //Found bigger sum
           if(sum > greatestSum || firstValid){
                  greatestSum = sum; 
                   firstValid = false; //Found first valid window (EDGE CASE)
           }
    }

    return greatestSum; 
}
