def max_valid_window_sum(nums, k):
	n = len(nums)
        window_sum = 0
        negatives = 0
        for i in range(k):
              window_sum += nums[i]
              if nums[i] < 0:
                   negatives += 1
         max_sum = window_sum if negatives <= 1 else None
         for i in range(k, n):
             window_sum += nums[i] - nums[i - k]
             if nums[i] < 0:
                 negatives += 1
             if nums[i - k] < 0:
                  negatives -= 1
             if negatives <= 1:
                  if max_sum is None or window_sum > max_sum:
                        max_sum = window_sum
         if max_sum is None:
              return 0
         return max_sum
