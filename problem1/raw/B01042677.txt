def max_valid_window_sum(nums, k):
   maxSum = 0
   windowSum = 0
   for n in nums[:k]:
      windowSum += k
   countNeg = 0
   for n in nums[:k]:
      if n < 0:
         countNeg += 1
   if countNeg <= 1:
      maxSum = windowSum

   for i in range(k, len(nums)):
      lastEl = i-k
      windowSum += nums[i] - nums[lastEl]
      if nums[i] < 0:
         countNeg += 1
      if nums[lastEl] < 0
         countNeg -= 1
      if countNeg <= 1:
         maxSum max(maxSum, windowSum)

   return maxSum
