def max_valid_window_sum(nums, k):
   	left = 0
   	window_sum = 0
   	neg_count = 0
   	max_sum = 0

   	for right in range(len(nums)):
  		window_sum += nums[right]
        	if nums[right] < 0:
            		neg_count += 1

        	if right - left + 1 > k:
            		if nums[left] < 0:
                		neg_count -= 1
            		window_sum -= nums[left]
            		left += 1

        	if right - left + 1 == k:
            		if neg_count <= 1:
                		max_sum = max(max_sum, window_sum)

    	return max_sum
