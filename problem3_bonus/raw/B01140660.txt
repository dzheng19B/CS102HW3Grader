def maxArea(self, height: List[int]) -> int:
        L = 0
        R = len(height) - 1
        area = 0

        while L < R:
            area = max(area, (R-L)*min(height[L], height[R]))
            if height[L] < height[R]:
                L+=1
            elif height[R] < height[L]:
                R-=1
            else:
                L+=1
                R-=1
        return area
#because of the two pointer L, R system, it is an O(n) algorithm. Because multiplication, and therefore area is commutative, we can multiply the area each time through the loop, and move through it depending on which height is bigger than the other. And if they're the same, we increment both.
