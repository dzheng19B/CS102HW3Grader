Def containsNearbyDuplicate(nums: List[int], k: int) -> bool:
	s = {}
	For i in range(len(nums)):
		if nums[i] in s:
			If(abs(i – s[nums[i]]) <= k):
				Return True
		s[i] = nums[i]
 	return False
