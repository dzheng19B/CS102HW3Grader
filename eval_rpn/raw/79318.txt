def evalRPN(arr) :

 	for i in range(len(arr)):

			stack.push(arr[i])
		if arr[i] == "*":
			term1 = stack[-1]
			stack.pop()

			stack.pop()
			term2 = int(term1)*int(term2)
			stack.push(term2)
		if arr[i] == "/":
			term1 = stack[-1]
			stack.pop()
			term2 = stack[-1]
			stack.pop()
			term2 = int(term2)/int(term1)

			term1 = stack[-1]
			stack.pop()
			term2 = stack[-1]
			stack.pop()
			term2 = int(term1)+int(term2)
			stack.push(term2)
		if arr[i] == "-":
			term1 = stack[-1]
			stack.pop()
			term2 = stack[-1]
			stack.pop()
			term2 = int(term2)-int(term1)
