def guessingGame(int n):
	lower = 1
	upper = n
	
	while(guess() != 0):
		middle = (upper + lower)/2
		if guess(middle) == 0:
			return middle
		
		if guess(middle) == -1:
			upper =  middle - 1
			
		if guess(middle) == 1:
			lower = middle + 1
