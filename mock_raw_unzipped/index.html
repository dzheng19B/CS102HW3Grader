def lengthOfLongestSubstring(s: str) -> int:
    left, max_len = 0, 0
    x = set()
    for i in range(len(s)):
        while s[i] in x:
            x.remove(s[left])
            left += 1
        x.add(s[i])
        max_len = max(max_len, i - left + 1)
    return max_len

def containsNearbyDuplicate(nums: List[int], k: int) -> bool:

#create a set since it cannot contain duplicate values, as i traverse through list[int] first, if that value is in set, then it is a duplicate, raising a flag

#if it is not seen in set, add it to set

dup = {}

for i, val in enumerate(nums):

if val in dup() and i - dup.val <= k:

return True

else:

dup[val] = i

return False

#test: [1,2,3,1], k = 3, dup{1:0, 2:1, 3:2} and then you see a duplicate value 1 and enter if statement

#time complexity: O(n), space complexity: O(n)

# check above comments for more details, traverse through nums and set new values to a set called dup, if val already in dup, and the difference of the indices are less than or equal to k, return True, otherwise return false
