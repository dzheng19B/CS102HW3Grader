def count_pairs(nums, T)
    l = 0
    r = 1
    numPairs = 0
    while l < len(nums):
        if nums[l] + nums[r] <= T:
            numPairs +=1
            r +=1
            if r == len(nums):
                 l = l + 1
                 r = l+1
        else:
           l+=1
           r+=1
